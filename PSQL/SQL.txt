CREATE TABLE jokes (
    id SERIAL PRIMARY KEY,
    text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    avg_rating NUMERIC(3,2) DEFAULT 0,
    ratings_count INT DEFAULT 0
);

CREATE TABLE tags (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) UNIQUE NOT NULL
);	


CREATE TABLE joke_tags (
    joke_id INT REFERENCES jokes(id) ON DELETE CASCADE,
    tag_id INT REFERENCES tags(id) ON DELETE CASCADE,
    PRIMARY KEY (joke_id, tag_id)
);

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    telegram_id BIGINT UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE ratings (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    joke_id INT REFERENCES jokes(id) ON DELETE CASCADE,
    value INT CHECK (value >= 1 AND value <= 5),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (user_id, joke_id)
);


CREATE INDEX jokes_text_idx 
ON jokes

CREATE INDEX idx_joke_tags_joke_id ON joke_tags(joke_id);
CREATE INDEX idx_jokes_rating ON jokes(avg_rating);

CREATE OR REPLACE VIEW jokes_with_rating AS
SELECT
    j.*,
    CASE
        WHEN j.source_rating IS NULL AND j.ratings_count = 0 THEN 0
        WHEN j.ratings_count = 0 THEN COALESCE(j.source_rating, 0)
        WHEN j.source_rating IS NULL THEN j.avg_rating
        ELSE (COALESCE(j.source_rating, 0) * 5 + j.avg_rating * j.ratings_count)
             / (5 + j.ratings_count)
    END AS final_rating
FROM jokes j;
